    12	  throw new Error("AUDIO_KEK must be a 64-character hex string (32 bytes)");

    16	/**
    17	 * Encrypts a DEK (Data Encryption Key) with the KEK
    18	 */
    19	function wrapKey(dek: Buffer): Buffer {
    20	  const iv = randomBytes(IV_LENGTH);
    21	  const cipher = createCipheriv(ALGORITHM, KEK, iv);
    22	  const encrypted = Buffer.concat([cipher.update(dek), cipher.final()]);
    23	  const tag = cipher.getAuthTag();
    24	  return Buffer.concat([iv, tag, encrypted]);
    25	}
    26	
    27	/**
    28	 * Decrypts a wrapped DEK using the KEK
    29	 */
    30	function unwrapKey(wrapped: Buffer): Buffer {
    31	  const iv = wrapped.slice(0, IV_LENGTH);
    32	  const tag = wrapped.slice(IV_LENGTH, IV_LENGTH + AUTH_TAG_LENGTH);
    33	  const data = wrapped.slice(IV_LENGTH + AUTH_TAG_LENGTH);
    34	
    35	  const decipher = createDecipheriv(ALGORITHM, KEK, iv);
    36	  decipher.setAuthTag(tag);
    37	
    38	  return Buffer.concat([decipher.update(data), decipher.final()]);
    39	}

    41	/**
    42	 * Encrypts raw PCM audio with per-file DEK
    43	 * Returns: { encryptedBuffer, encryptedKey (hex) }
    44	 */
    45	export async function encryptAudio(
    46	  pcm: Float32Array
    47	): Promise<{ encryptedBuffer: Buffer; encryptedKey: string }> {
    48	  const dek = randomBytes(KEY_LENGTH);
    49	  const iv = randomBytes(IV_LENGTH);
    50	
    51	  const raw = Buffer.from(pcm.buffer, pcm.byteOffset, pcm.byteLength);
    52	
    53	  const cipher = createCipheriv(ALGORITHM, dek, iv);
    54	  const encrypted = Buffer.concat([cipher.update(raw), cipher.final()]);
    55	  const tag = cipher.getAuthTag();
    56	
    57	  const encryptedBuffer = Buffer.concat([iv, tag, encrypted]);
    58	  const wrapped = wrapKey(dek);
    59	  const encryptedKey = wrapped.toString("hex");
    60	
    61	  logger.debug({ size: raw.length, encryptedSize: encryptedBuffer.length }, "Audio encrypted");
    62	
    63	  return { encryptedBuffer, encryptedKey };
    64	}

    12	async function getClient() {
    13	  if (client) return client;
    14	
    15	  const packageDefinition = await protoLoader.load(PROTO_PATH, {
    16	    keepCase: true,
    17	    longs: String,
    18	    enums: String,
    19	    defaults: true,
    20	    oneofs: true,
    21	  });
    22	
    23	  const proto = grpc.loadPackageDefinition(packageDefinition) as any;
    24	
    25	  client = new proto.inference.InferenceService(
    26	    GATEWAY_URL,
    27	    grpc.credentials.createInsecure(),
    28	    {
    29	      "grpc.max_receive_message_length": 100 * 1024 * 1024, // 100MB
    30	      "grpc.max_send_message_length": 100 * 1024 * 1024,
    31	    }
    32	  );
    33	
    34	  return client;
    35	}

    37	export async function callInferenceGateway<T>(
    38	  method: string,
    39	  request: any,
    40	  /** ACR-E6-001 (P2) — ADDITIVE options: enforced deadline + tenant metadata.
    41	   * Omitted = exact legacy behavior (existing STT/Align/G2P callers untouched). */
    42	  options?: { deadlineMs?: number; tenantId?: string }
    43	): Promise<T> {
    44	  const startTime = Date.now();
    45	  const log = logger.child({ method, correlationId: request.correlationId });
    46	
    47	  try {
    48	    const c = await getClient();
    49	
    50	    return new Promise((resolve, reject) => {
    51	      const metadata = new grpc.Metadata();
    52	      metadata.add("authorization", `Bearer ${API_KEY}`);
    53	      metadata.add("x-correlation-id", request.correlationId || "");
    54	      if (options?.tenantId) metadata.add("x-tenant-id", options.tenantId);
    55	
    56	      // grpc-js: (request, metadata, [options], callback). The options form
    57	      // (with an enforced deadline) is used ONLY when a deadline is provided.
    58	      const callArgs: unknown[] = [request, metadata];
    59	      if (options?.deadlineMs && options.deadlineMs > 0) {
    60	        callArgs.push({ deadline: Date.now() + options.deadlineMs });
    61	      }
    62	      callArgs.push((err: any, response: any) => {
    63	        const duration = (Date.now() - startTime) / 1000;
    64	
    65	        metrics.modelLatency
    66	          .labels(method, "latest")
    67	          .observe(duration);
    68	
    69	        if (err) {
    70	          log.error({ err, duration }, "Inference call failed");
    71	          metrics.pipelineErrors
    72	            .labels(method, err.code || "UNKNOWN")
    73	            .inc();
    74	          reject(err);
    75	        } else {
    76	          log.info({ duration }, "Inference call succeeded");
    77	          resolve(response);
    78	        }
    79	      });
    80	      (c[method] as (...a: unknown[]) => void)(...callArgs);
    81	    });
    82	  } catch (err) {
    83	    log.error({ err }, "Failed to call inference gateway");
    84	    throw err;
    85	  }
    86	}

     8	/**
     9	 * Speech-to-Text using Whisper Large V3
    10	 *
    11	 * Returns word-level timestamps + real confidence scores from Whisper signals:
    12	 *   - no_speech_prob
    13	 *   - avg_logprob
    14	 *   - compression_ratio
    15	 */
    16	@injectable()
    17	export class WhisperSTT {
    18	  async transcribe(
    19	    audio: Float32Array,
    20	    sampleRate: number,
    21	    correlationId?: string
    22	  ): Promise<STTResult> {
    23	    const log = logger.child({ component: "WhisperSTT", correlationId });
    24	    const startTime = Date.now();
    25	
    26	    const { whisper: whisperCfg } = modelsConfig;
    27	
    28	    try {
    29	      const response = await callInferenceGateway<any>("Transcribe", {
    30	        audio: Array.from(audio),
    31	        sample_rate: sampleRate,
    32	        language: whisperCfg.parameters.language,
    33	        task: whisperCfg.parameters.task,
    34	        word_timestamps: whisperCfg.parameters.word_timestamps,
    35	        chunk_length_s: whisperCfg.parameters.chunk_length_s,
    36	        stride_length_s: whisperCfg.parameters.stride_length_s,
    37	        correlation_id: correlationId,
    38	      });
    39	
    40	      const words: WordTimestamp[] = (response.segments || []).flatMap((seg: any) =>
    41	        (seg.words || []).map((w: any) => ({
    42	          word: w.word.trim(),
    43	          start: w.start,
    44	          end: w.end,
    45	          confidence: this.combineConfidence(
    46	            w.no_speech_prob ?? seg.no_speech_prob ?? 0.5,
    47	            w.avg_logprob ?? seg.avg_logprob ?? -0.5,
    48	            seg.compression_ratio ?? 1.0
    49	          ),
    50	          noSpeechProb: w.no_speech_prob ?? seg.no_speech_prob ?? 0,
    51	          avgLogprob: w.avg_logprob ?? seg.avg_logprob ?? 0,
    52	        }))
    53	      );
    54	
    55	      const result: STTResult = {
    56	        text: response.text.trim(),
    57	        words,
    58	        language: "ar",
    59	        engine: "whisper-large-v3",
    60	        modelVersion: whisperCfg.version,
    61	      };
    62	
    63	      const duration = (Date.now() - startTime) / 1000;
    64	      metrics.pipelineDuration.labels("stt").observe(duration);
    65	
    66	      log.info({ duration, words: words.length }, "STT completed");
    67	
    68	      return result;
    69	    } catch (err) {
    70	      log.error({ err }, "STT failed");
    71	      throw err;
    72	    }
    73	  }
    74	
    75	  /**
    76	   * Combine Whisper's 3 confidence signals into one 0..1 score
    77	   */
    78	  private combineConfidence(noSpeech: number, logprob: number, compression: number): number {
    79	    const speechScore = 1 - noSpeech;
    80	    const logprobScore = Math.max(0, Math.min(1, (logprob + 1)));
    81	    const compressionScore = compression > 2.4 ? 0.3 : 1.0;
    82	
    83	    return 0.5 * speechScore + 0.3 * logprobScore + 0.2 * compressionScore;
    84	  }
    85	}

     7	/**
     8	 * Audio Enhancement using DeepFilterNet (native, not WASM)
     9	 *
    10	 * Pipeline:
    11	 *   PCM 16k → DeepFilterNet subprocess → PCM 16k (cleaned)
    12	 */
    13	@injectable()
    14	export class AudioEnhancement {
    15	  private workerPath: string;
    16	
    17	  constructor() {
    18	    this.workerPath = process.env.DFN_WORKER_PATH || "./dfn-worker";
    19	  }
    20	
    21	  async denoise(pcm16k: Float32Array, correlationId?: string): Promise<Float32Array> {
    22	    const log = logger.child({ component: "AudioEnhancement", correlationId });
    23	    const startTime = Date.now();
    24	
    25	    return new Promise((resolve, reject) => {
    26	      const child = spawn(this.workerPath, [
    27	        "--sample-rate", String(audioConfig.input.sampleRate),
    28	        "--format", "f32le",
    29	      ], { stdio: ["pipe", "pipe", "pipe"] });
    30	
    31	      const chunks: Buffer[] = [];
    32	      let stderr = "";
    33	
    34	      child.stdout.on("data", (d: Buffer) => chunks.push(d));
    35	      child.stderr.on("data", (d: Buffer) => {
    36	        stderr += d.toString();
    37	        log.debug({ stderr: d.toString() }, "DeepFilterNet output");
    38	      });
    39	
    40	      child.on("close", (code) => {
    41	        const duration = (Date.now() - startTime) / 1000;
    42	        metrics.pipelineDuration.labels("audio_enhancement").observe(duration);
    43	
    44	        if (code !== 0) {
    45	          log.error({ code, stderr, duration }, "DeepFilterNet failed");
    46	          reject(new Error(`DeepFilterNet exited with code ${code}: ${stderr}`));
    47	          return;
    48	        }
    49	
    50	        const buf = Buffer.concat(chunks);
    51	        const result = new Float32Array(buf.buffer, buf.byteOffset, buf.byteLength / 4);
    52	
    53	        log.info({ duration, outputSamples: result.length }, "Audio enhancement completed");
    54	        resolve(result);
    55	      });
    56	
    57	      child.on("error", (err) => {
    58	        log.error({ err }, "Failed to spawn DeepFilterNet");
    59	        reject(err);
    60	      });
    61	
    62	      // Send raw PCM
    63	      const inputBuf = Buffer.from(pcm16k.buffer, pcm16k.byteOffset, pcm16k.byteLength * 4);
    64	      child.stdin.write(inputBuf);
    65	      child.stdin.end();
    66	    });
    67	  }
    68	
    69	  /**
    70	   * Validate audio buffer
    71	   */
    72	  validate(pcm: Float32Array, sampleRate: number): void {
    73	    if (pcm.length === 0) {
    74	      throw new Error("Audio buffer is empty");
    75	    }
    76	
    77	    const durationSec = pcm.length / sampleRate;
    78	    if (durationSec < audioConfig.processing.minDurationSec) {
    79	      throw new Error(`Audio too short: ${durationSec}s < ${audioConfig.processing.minDurationSec}s`);
    80	    }
    81	
    82	    if (durationSec > audioConfig.processing.maxDurationSec) {
    83	      throw new Error(`Audio too long: ${durationSec}s > ${audioConfig.processing.maxDurationSec}s`);
    84	    }
    85	
    86	    if (sampleRate !== audioConfig.input.sampleRate) {
    87	      throw new Error(`Invalid sample rate: ${sampleRate} != ${audioConfig.input.sampleRate}`);
    88	    }
    89	  }
    90	}
